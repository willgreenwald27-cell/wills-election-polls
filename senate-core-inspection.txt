function shortLastName(name){
  const full=String(name||'').trim();
  if(!full) return 'Leader';
  const parts=full.split(/\s+/);
  if(parts.length===1) return parts[0];
  const particles=new Set(['van','von','de','del','di','da','la','le','st.','st','du']);
  const last=parts[parts.length-1];
  const prev=parts[parts.length-2];
  if(particles.has(prev.toLowerCase())) return prev + ' ' + last;
  return last;
}

function fmtMarginNumber(n){
  const rounded=Math.round(n*10)/10;
  return rounded.toFixed(1);
}

function statePollMarginClass(s){
  const ordered=orderedStateCandidates(s);
  const lead=ordered[0];
  const leadPoll=pollValue(lead.poll), trailPoll=pollValue((ordered[1]||{}).poll);
  if(leadPoll===null || trailPoll===null || Math.abs(leadPoll-trailPoll)<0.05) return 'even';
  return candidatePollClass(lead.name,lead.party);
}

function statePollMarginLabel(s){
  const ordered=orderedStateCandidates(s);
  const lead=ordered[0], trail=ordered[1];
  const leadPoll=pollValue(lead.poll), trailPoll=pollValue(trail.poll);
  if(leadPoll===null || trailPoll===null) return 'No avg';
  const diff=Math.round((leadPoll-trailPoll)*10)/10;
  if(Math.abs(diff)<0.05) return 'Even';
  return `${shortLastName(lead.name)} +${fmtMarginNumber(diff)}`;
}

function pollCandidateParty(p,name,slot){
  const s=stateData[p.state]||{};
  const n=String(name||'').trim().toLowerCase();
  if(n && n===String(s.candidate1||'').trim().toLowerCase()) return s.candidate1Party||'';
  if(n && n===String(s.candidate2||'').trim().toLowerCase()) return s.candidate2Party||'';
  return slot===1?(s.candidate1Party||''):(s.candidate2Party||'');
}
function orderedPollCandidates(p){
  const candidates=[
    {slot:1,name:p.c1||'Candidate 1',poll:p.c1Pct,party:pollCandidateParty(p,p.c1,1)},
    {slot:2,name:p.c2||'Candidate 2',poll:p.c2Pct,party:pollCandidateParty(p,p.c2,2)}
  ];
  return candidates.sort((a,b)=>{
    const av=pollValue(a.poll), bv=pollValue(b.poll);
    if(av===null && bv===null) return a.slot-b.slot;
    if(av===null) return 1;
    if(bv===null) return -1;
    if(bv!==av) return bv-av;
    return a.slot-b.slot;
  });
}

function pollValue(v){
  const n=Number(v);
  return Number.isFinite(n) && String(v).trim()!=='' ? n : null;
}
function orderedStateCandidates(s){
  const candidates=[
    {slot:1,name:s.candidate1||'Candidate 1',party:s.candidate1Party||'',poll:s.candidate1Poll,odds:s.candidate1Odds},
    {slot:2,name:s.candidate2||'Candidate 2',party:s.candidate2Party||'',poll:s.candidate2Poll,odds:s.candidate2Odds}
  ];
  return candidates.sort((a,b)=>{
    const av=pollValue(a.poll), bv=pollValue(b.poll);
    if(av===null && bv===null) return a.slot-b.slot;
    if(av===null) return 1;
    if(bv===null) return -1;
    if(bv!==av) return bv-av;
    return a.slot-b.slot;
  });
}


function shortLastName(name){
  const full=String(name||'').trim();
  if(!full) return 'Leader';
  const parts=full.split(/\s+/);
  if(parts.length===1) return parts[0];
  const particles=new Set(['van','von','de','del','di','da','la','le','st.','st','du']);
  const last=parts[parts.length-1];
  const prev=parts[parts.length-2];
  if(particles.has(prev.toLowerCase())) return prev + ' ' + last;
  return last;
}

function fmtMarginNumber(n){
  const rounded=Math.round(n*10)/10;
  return rounded.toFixed(1);
}

function statePollMarginClass(s){
  const ordered=orderedStateCandidates(s);
  const lead=ordered[0];
  const leadPoll=pollValue(lead.poll), trailPoll=pollValue((ordered[1]||{}).poll);
  if(leadPoll===null || trailPoll===null || Math.abs(leadPoll-trailPoll)<0.05) return 'even';
  return candidatePollClass(lead.name,lead.party);
}

function statePollMarginLabel(s){
  const ordered=orderedStateCandidates(s);
  const lead=ordered[0], trail=ordered[1];
  const leadPoll=pollValue(lead.poll), trailPoll=pollValue(trail.poll);
  if(leadPoll===null || trailPoll===null) return 'No avg';
  const diff=Math.round((leadPoll-trailPoll)*10)/10;
  if(Math.abs(diff)<0.05) return 'Even';
  return `${shortLastName(lead.name)} +${fmtMarginNumber(diff)}`;
}

  const p1=errNum(x.poll1),p2=errNum(x.poll2),a1=errNum(x.actual1),a2=errNum(x.actual2);

  return ps!==0&&as!==0&&ps!==as;
}

}

function positionHoverCard(ev){
  const wrap=document.querySelector('#page-senate .map-wrap');
  const card=document.getElementById('mapHoverCard');
  if(!wrap||!card) return;
  const rect=wrap.getBoundingClientRect();
  let x=ev.clientX-rect.left+16, y=ev.clientY-rect.top+16;
  const maxX=wrap.clientWidth-card.offsetWidth-10;
  const maxY=wrap.clientHeight-card.offsetHeight-10;
  if(x>maxX) x=Math.max(10,ev.clientX-rect.left-card.offsetWidth-16);
  if(y>maxY) y=Math.max(10,ev.clientY-rect.top-card.offsetHeight-16);
  card.style.left=Math.max(10,x)+'px';
  card.style.top=Math.max(10,y)+'px';
}
